# Mac apps (Homebrew casks) that are optional on work machines.
# Personal machines install all of them. On a work machine,
# scripts/setup-machine.sh asks which ones to install and saves the picks in
# ~/.config/dotfiles/optional-casks, which nix/darwin.nix reads.
# Format: <cask> # <description shown in the pick-list>

# Security & Password Management
1password           # Password manager
1password-cli       # Password manager CLI
tailscale-app       # Mesh VPN
nordvpn             # VPN
ausweisapp          # German ID card authentication

# AI assistants
claude              # Claude desktop app
chatgpt             # ChatGPT desktop app
antigravity-cli     # Google's AI coding CLI

# Productivity & Utilities
vorssaint           # Menu bar toolkit with a Dynamic Island
google-chrome       # Web browser
firefox             # Web browser
handy               # Speech recognition
openwhispr          # Dictation and meeting notes, local models (open-source)
shottr              # Screenshot tool
wordcraft           # Word processor (open-source MS Word alternative)
stirling-pdf        # Local PDF toolkit (merge, split, PDF to image, OCR)

# Communication
slack               # Team chat
whatsapp            # Messaging

# Media & Creative
blender             # 3D creation suite
comfy               # AI image, 3D and sound generator (ComfyUI)
spotify             # Music
